// =====================================================================
// Radiant Silvergun Options -- one top-level menu, branching by concern
// rather than one flat submenu. Weapon Fidelity Options holds the FX
// toggles (set-agnostic, apply to every weapon set); Vanilla+ Weapon
// Behavior holds settings specific to that one set. Splitting it this
// way means a Vanilla+-only setting doesn't have to sit next to unrelated
// FX toggles just because both happened to exist at the same time --
// each branch grows independently as more settings get added.
// =====================================================================
AddOptionMenu "OptionsMenu"
{
	StaticText " "
	Submenu "Radiant Silvergun Options", "RS_MainOptions"
}

OptionMenu "RS_MainOptions"
{
	Title "Radiant Silvergun Options"
	StaticText " "

	Submenu "Weapon Fidelity Options", "RS_FXSettings"
	Submenu "Vanilla+ Weapon Behavior", "RS_VanillaPlusSettings"
	Submenu "Dual Class Behavior", "RS_DualClassSettings"
	Submenu "$TFLV_OPT_TITLE", "GunBonsaiOptions"
	Submenu "Kill Rewards", "RS_BitsSettings"
	StaticText " "
	Submenu "Debug / Testing", "RS_DebugMenu"
}
